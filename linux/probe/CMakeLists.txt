cmake_minimum_required(VERSION 3.25)
project(ea_linux_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(rexglue 0.10.0 EXACT REQUIRED CONFIG)
find_package(ZLIB REQUIRED)
add_executable(ea_linux_probe ../../tools/graphics-probe/diagnostics.cpp ../../src/frame_capture.cpp)
target_compile_definitions(ea_linux_probe PRIVATE REX_HAS_FIDELITYFX_SDK=1)
target_link_libraries(ea_linux_probe PRIVATE rex::runtime ZLIB::ZLIB)
rexglue_apply_target_settings(ea_linux_probe)
add_executable(ea_vulkan_probe vulkan.cpp)
target_compile_definitions(ea_vulkan_probe PRIVATE REX_HAS_FIDELITYFX_SDK=1)
target_link_libraries(ea_vulkan_probe PRIVATE rex::runtime dl)
rexglue_apply_target_settings(ea_vulkan_probe)
