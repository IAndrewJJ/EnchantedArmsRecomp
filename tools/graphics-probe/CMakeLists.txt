cmake_minimum_required(VERSION 3.25)
project(ea_graphics_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(rexglue 0.10.0 EXACT REQUIRED CONFIG)
add_executable(ea_sdk_link_probe sdk_link.cpp)
target_link_libraries(ea_sdk_link_probe PRIVATE rex::runtime)
rexglue_apply_target_settings(ea_sdk_link_probe)
add_executable(ea_graphics_probe probe.cpp)
target_compile_definitions(ea_graphics_probe PRIVATE REX_HAS_FIDELITYFX_SDK=1)
target_link_libraries(ea_graphics_probe PRIVATE rex::runtime)
rexglue_apply_target_settings(ea_graphics_probe)
add_executable(ea_pacing_probe pacing.cpp)
add_executable(ea_timing_probe timing.cpp)
add_executable(ea_fps_counter_probe fps_counter.cpp ../../src/fps_counter.cpp)
target_compile_definitions(ea_fps_counter_probe PRIVATE REX_HAS_FIDELITYFX_SDK=1)
target_link_libraries(ea_fps_counter_probe PRIVATE rex::runtime)
rexglue_apply_target_settings(ea_fps_counter_probe)
add_executable(ea_diagnostics_probe diagnostics.cpp ../../src/frame_capture.cpp)
target_compile_definitions(ea_diagnostics_probe PRIVATE REX_HAS_FIDELITYFX_SDK=1)
target_link_libraries(ea_diagnostics_probe PRIVATE rex::runtime windowscodecs ole32)
rexglue_apply_target_settings(ea_diagnostics_probe)
